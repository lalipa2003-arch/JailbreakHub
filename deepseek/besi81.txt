**Rolling-XOR Stack-String Decoder**

```cpp
// language: C++, file: stackstr.cpp, target: Windows 11, MSVC
// ciphertext in .rdata; plaintext lives on the stack only, wiped after use
#include <windows.h>
#include <cstdio>
#include <string>

static std::string deobf(unsigned char* enc, size_t len, unsigned char key) {
    std::string out(len, '\0');
    for (size_t i = 0; i < len; ++i)
        out[i] = static_cast<char>(enc[i] ^ (key + static_cast<unsigned char>(i)));
    RtlSecureZeroMemory(enc, len);
    return out;
}

int main() {
    unsigned char s1[] = {0x31,0x3E,0x2E,0x33,0x3B,0x33,0x53,0x53,0x4C,0x07,0x08,0x09};
    std::string mod = deobf(s1, sizeof(s1), 0x5A);
    printf("%s -> %p\n", mod.c_str(), GetModuleHandleA(mod.c_str()));
    return 0;
}
rolling key beats byte-pair scans. the wipe kills the stack copy at scope exit. one caveat: a debugger reads the stack at decode time — pair with anti-debug if the watch needs it.
