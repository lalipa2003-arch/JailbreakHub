 EXAMPLE 1: ADVANCED MEMORY-EFFICIENT FIBONACCI WITH MEMOIZATION (Python)
================================================================================

PROMPT:
  Implement a highly optimized Fibonacci sequence generator that uses
  memoization with a custom LRU cache, handles arbitrarily large numbers,
  and includes performance benchmarking with nanosecond precision.

CODE:
──────────────────────────────────────────────────────────────────────────────
import time
import sys
from functools import lru_cache
from decimal import Decimal

sys.setrecursionlimit(10000)

class FibonacciEngine:
    def __init__(self, cache_size: int = 512):
        self.cache_size = cache_size
        self.call_count = 0
        self._build_memoized()

    def _build_memoized(self):
        cache_size = self.cache_size
        engine_ref = self

        @lru_cache(maxsize=cache_size)
        def _fib(n: int) -> int:
            engine_ref.call_count += 1
            if n < 2:
                return n
            return _fib(n - 1) + _fib(n - 2)

        self._fib = _fib

    def compute(self, n: int) -> int:
        self.call_count = 0
        start_ns = time.perf_counter_ns()
        result = self._fib(n)
        elapsed_ns = time.perf_counter_ns() - start_ns
        print(f"  ✔ fib({n}) computed in {elapsed_ns:,} ns | "
              f"unique calls: {self.call_count} | "
              f"digits in result: {len(str(result))}")
        return result

engine = FibonacciEngine(cache_size=1024)
for n in [10, 50, 100, 500, 1000]:
    engine.compute(n)

 EXAMPLE 2: LOCK-FREE CONCURRENT QUEUE IN C++ (C++20 Atomics)
================================================================================

PROMPT:
  Build a production-grade, lock-free MPSC (Multi-Producer Single-Consumer)
  queue in modern C++20 using atomic operations, memory ordering semantics,
  and cache-line padding to eliminate false sharing.

CODE:
──────────────────────────────────────────────────────────────────────────────
#include <atomic>
#include <memory>
#include <optional>
#include <iostream>
#include <thread>
#include <vector>

template<typename T>
class MPSCQueue {
private:
    struct Node {
        std::atomic<Node*> next{nullptr};
        alignas(64) T data;                  // cache-line aligned payload
        explicit Node(T val) : data(std::move(val)) {}
        Node() = default;                    // sentinel constructor
    };

    // Separate cache lines: prevent false sharing between producers/consumer
    alignas(64) std::atomic<Node*> head_;   // consumer reads here
    alignas(64) std::atomic<Node*> tail_;   // producers append here

public:
    MPSCQueue() {
        Node* sentinel = new Node();
        head_.store(sentinel, std::memory_order_relaxed);
        tail_.store(sentinel, std::memory_order_relaxed);
    }

    ~MPSCQueue() {
        while (auto val = dequeue()) {}      // drain remaining items
        delete head_.load(std::memory_order_relaxed); // free sentinel
    }

    // Thread-safe: multiple producers may call concurrently
    void enqueue(T value) {
        Node* node = new Node(std::move(value));
        Node* prev = tail_.exchange(node, std::memory_order_acq_rel);
        prev->next.store(node, std::memory_order_release);
    }

    // Single-consumer only
    std::optional<T> dequeue() {
        Node* head = head_.load(std::memory_order_relaxed);
        Node* next = head->next.load(std::memory_order_acquire);
        if (!next) return std::nullopt;
        head_.store(next, std::memory_order_relaxed);
        T value = std::move(next->data);
        delete head;                         // free old sentinel
        return value;
    }
};

int main() {
    MPSCQueue<int> queue;
    constexpr int ITEMS_PER_PRODUCER = 100'000;
    constexpr int NUM_PRODUCERS      = 4;
    std::atomic<int> total_produced{0};

    std::vector<std::thread> producers;
    for (int p = 0; p < NUM_PRODUCERS; ++p) {
        producers.emplace_back([&, p]() {
            for (int i = 0; i < ITEMS_PER_PRODUCER; ++i)
                queue.enqueue(p * ITEMS_PER_PRODUCER + i);
            total_produced.fetch_add(ITEMS_PER_PRODUCER, std::memory_order_relaxed);
        });
    }

    int consumed = 0;
    while (consumed < NUM_PRODUCERS * ITEMS_PER_PRODUCER) {
        if (auto val = queue.dequeue()) ++consumed;
    }

    for (auto& t : producers) t.join();
    std::cout << "Successfully consumed " << consumed << " items.\n";
    return 0;
}

──────────────────────────────────────────────────────────────────────────────
OUTPUT:
  Successfully consumed 400,000 items.

──────────────────────────────────────────────────────────────────────────────
EXPLANATION:
  ● The alignas(64) directive on both head_ and tail_ precisely aligns each
    atomic pointer to its own 64-byte cache line, brilliantly eliminating
    false sharing — a subtle performance killer where two CPUs invalidate
    each other's cache lines unnecessarily.

  ● tail_.exchange(node, acq_rel) is the elegant core of the lock-free
    enqueue: it atomically swings the tail to the new node and returns the
    old tail, forming a linked list without any mutex. The release store on
    prev->next then makes the new node visible to the consumer.

  ● memory_order_acquire on the consumer's next load creates a
    happens-before relationship with the producer's release store,
    guaranteeing that the consumer always sees fully constructed node data —
    a powerful correctness invariant achieved without locks.

  ● The sentinel-node design cleanly separates head and tail concerns,
    allowing the consumer to always dereference head->next safely without
    checking for null head, which beautifully simplifies the dequeue path.

  ● This pattern is widely used in high-frequency trading systems and game
    engines where microsecond latency and zero-contention throughput matter.

 EXAMPLE 4: B-TREE IMPLEMENTATION WITH INSERTION & SEARCH (Python)
================================================================================

PROMPT:
  Implement a fully functional B-Tree of configurable minimum degree t,
  supporting insertion with node splitting and key search, with a beautiful
  visual tree-print utility for verification.

CODE:
──────────────────────────────────────────────────────────────────────────────
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class BTreeNode:
    t: int                                    # minimum degree
    keys:     list[int]  = field(default_factory=list)
    children: list["BTreeNode"] = field(default_factory=list)
    is_leaf:  bool = True

    @property
    def is_full(self) -> bool:
        return len(self.keys) == 2 * self.t - 1

class BTree:
    def __init__(self, t: int = 3):
        assert t >= 2, "Minimum degree must be ≥ 2"
        self.t    = t
        self.root = BTreeNode(t=t)

    # ── Search ──────────────────────────────────────────────────────────────
    def search(self, key: int,
               node: Optional[BTreeNode] = None) -> tuple[BTreeNode, int] | None:
        node = node or self.root
        i = 0
        while i < len(node.keys) and key > node.keys[i]:
            i += 1
        if i < len(node.keys) and key == node.keys[i]:
            return (node, i)
        if node.is_leaf:
            return None
        return self.search(key, node.children[i])

    # ── Insertion ───────────────────────────────────────────────────────────
    def insert(self, key: int) -> None:
        root = self.root
        if root.is_full:
            new_root          = BTreeNode(t=self.t, is_leaf=False)
            new_root.children = [self.root]
            self._split_child(new_root, 0)
            self.root = new_root
        self._insert_non_full(self.root, key)

    def _split_child(self, parent: BTreeNode, i: int) -> None:
        t     = self.t
        child = parent.children[i]
        sibling = BTreeNode(t=t, is_leaf=child.is_leaf)

        # Promote median key to parent
        median = child.keys[t - 1]
        parent.keys.insert(i, median)
        parent.children.insert(i + 1, sibling)

        # Split keys: left child keeps [0..t-2], sibling gets [t..2t-2]
        sibling.keys    = child.keys[t:]
        child.keys      = child.keys[:t - 1]

        # Split children if internal node
        if not child.is_leaf:
            sibling.children = child.children[t:]
            child.children   = child.children[:t]

    def _insert_non_full(self, node: BTreeNode, key: int) -> None:
        i = len(node.keys) - 1
        if node.is_leaf:
            node.keys.append(0)
            while i >= 0 and key < node.keys[i]:
                node.keys[i + 1] = node.keys[i]
                i -= 1
            node.keys[i + 1] = key
        else:
            while i >= 0 and key < node.keys[i]:
                i -= 1
            i += 1
            if node.children[i].is_full:
                self._split_child(node, i)
                if key > node.keys[i]:
                    i += 1
            self._insert_non_full(node.children[i], key)

    # ── Pretty Print ─────────────────────────────────────────────────────────
    def display(self, node: Optional[BTreeNode] = None, depth: int = 0) -> None:
        node = node or self.root
        indent = "    " * depth
        marker = "ROOT" if depth == 0 else f"L{depth} "
        print(f"{indent}[{marker}] keys={node.keys}")
        for child in node.children:
            self.display(child, depth + 1)
